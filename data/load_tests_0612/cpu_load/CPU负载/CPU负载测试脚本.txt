#!/bin/bash

# 配置参数
SERVER_IP="192.168.100.100"  # PTP服务器IP
TEST_DURATION=1200           # 测试时长20分钟(1200秒)
CPU_CORES=(1 2 3 4)          # 测试的CPU核心数
LOG_DIR="/root/collected_sync_logs"
PTPATR_CMD="ptpatr -c $SERVER_IP -x"
ORIGINAL_LOG="/root/tool/log0-original.log"
STRESS_DURATION=$((TEST_DURATION + 10))  # stress运行时间稍长于测试时长

# 确保日志目录存在
mkdir -p "$LOG_DIR"

# 函数: 记录CPU温度
monitor_temp() {
  local log_file="$1"
  local start_time=$(date +%s)
  
  echo "时间(秒) 温度(°C)" > "$log_file"
  
  while true; do
    current_time=$(date +%s)
    elapsed_time=$((current_time - start_time))
    temp=$(cat /sys/class/thermal/thermal_zone0/temp)
    temp_c=$(echo "scale=2; $temp / 1000.0" | bc)
    echo "$elapsed_time $temp_c" >> "$log_file"
    sleep 1
  done
}

# 主测试循环
for cores in "${CPU_CORES[@]}"; do
  echo "开始测试CPU负载: $cores 个核心"
  TEST_START_TIME=$(date +"%Y%m%d_%H%M%S")
  TEMP_LOG="${LOG_DIR}/CPU${cores}_temp.log"
  PTP_LOG="${LOG_DIR}/log0-original_cpu${cores}.log"
  
  # 清理旧日志文件
  [ -f "$ORIGINAL_LOG" ] && rm "$ORIGINAL_LOG"
  [ -f "$TEMP_LOG" ] && rm "$TEMP_LOG"
  
  # 启动温度监控
  monitor_temp "$TEMP_LOG" &
  TEMP_PID=$!
  
  # 启动PTP测试
  echo "启动PTP时钟同步测试..."
  $PTPATR_CMD > /dev/null 2>&1 &
  PTP_PID=$!
  
  # 启动CPU压力测试
  echo "启动stress测试: $cores 个CPU核心..."
  stress --cpu $cores --timeout $STRESS_DURATION > /dev/null 2>&1 &
  STRESS_PID=$!
  
  # 显示测试进度
  echo "测试进行中 (CPU核心: $cores, 持续时间: 20分钟)..."
  sleep $TEST_DURATION
  
  # 等待日志生成
  sleep 5
  
  # 处理PTP日志
  if [ -f "$ORIGINAL_LOG" ]; then
    mv "$ORIGINAL_LOG" "$PTP_LOG"
    echo "PTP日志已保存为: $PTP_LOG"
  else
    echo "警告: 未找到PTP日志文件 $ORIGINAL_LOG"
  fi
  
  # 清理进程
  echo "测试完成，清理进程..."
  kill $PTP_PID $STRESS_PID $TEMP_PID 2>/dev/null
  wait $PTP_PID $STRESS_PID $TEMP_PID 2>/dev/null
  
  echo "CPU $cores 核心测试完成"
  echo "----------------------------------------"
done

echo "所有CPU负载测试完成!"
echo "日志文件保存在: $LOG_DIR"
echo "包括:"
echo "- PTP日志: log0-original_cpu[1-4].log"
echo "- 温度日志: CPU[1-4]_temp.log"
